# apt packages installed on every machine (core tier).
# gopass and gh come from their official apt repos (registered by bootstrap.sh
# via add_apt_repo). Ubuntu 26.04 dropped gopass from universe, and its gh 2.46
# package lacks features used by ~/.scripts/deploy_secrets/gh_tokens.sh.
bat # yazi previews call `bat`; Ubuntu ships it as batcat, bootstrap.sh adds the shim
btop # system monitor
bubblewrap # bwrap: Claude Code's Linux sandbox
build-essential # cc/make: nvim-treesitter compiles parsers, cargo builds
ca-certificates
curl
eza # modern ls (plain universe pkg; old deb.gierens.de repo dropped)
fastfetch # system info (plain universe pkg; old PPA dropped)
fd-find # fd: LazyVim file finding; Ubuntu ships it as fdfind, bootstrap.sh adds the shim
fontconfig # fc-cache: register Ubuntu Mono Nerd Font after its system-wide install
fonts-noto-core # broad Unicode font coverage
fuse3 # fusermount3 for sshfs mounts (~/.scripts/hetzner_mount)
fzf # fuzzy finder: zshrc keybindings + yazi's fzf/fg plugins
gh # GitHub CLI; official cli.github.com apt repo
git
git-lfs # ~/.gitconfig declares the lfs filter with required=true
gnupg # gpg: gopass encryption + personal key import (~/.scripts/gpg/import-gpg-key.sh)
scdaemon # gpg-agent smartcard support required to decrypt the personal gopass store
gopass
jq
libclang-dev # bindgen needs libclang to build tree-sitter-cli (rquickjs-sys) in lib/install-rust.sh
libsqlite3-dev # LazyVim: sqlite-backed plugins build against it
luarocks # LazyVim rocks support
mosh
npm # node for LazyVim: copilot.lua runtime; mason installs pyright/prettierd/markdownlint-cli2/json+yaml LSPs via npm
openssh-server
python3-venv # mason python tooling / venvs
restic # ~/.scripts/restic_b2_backups runtime dep
ripgrep # rg: yazi's fg plugin content search
sqlite3
sshfs # ~/.scripts/hetzner_mount runtime dep
tmux
vim-gtk3 # vim with +clipboard (needed by ~/.vimrc's clipboard=unnamedplus)
xz-utils # xz: extract the Ubuntu Mono Nerd Font release archive
zoxide # z/zi commands + interactive-only cd wrapper + yazi's zoxide plugin
zsh
